import requests

# --- CẤU HÌNH ---
# Lấy API Key từ: https://www.firecrawl.dev/app/api-keys
API_KEY = ""  # Thay bằng key thật của bạn
BASE_URL = "https://api.firecrawl.dev/v1/scrape"

def test_firecrawl():
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json"
    }
    
    # Dữ liệu test: Thử scrape trang chủ Google hoặc chính trang Firecrawl
    payload = {
        "url": "https://www.google.com",
        "formats": ["markdown"]
    }

    print(f"🚀 Đang gửi yêu cầu tới Firecrawl...")

    try:
        response = requests.post(BASE_URL, headers=headers, json=payload, timeout=20)
        
        if response.status_code == 200:
            print("✅ Kết nối thành công!")
            data = response.json()
            if data.get("success"):
                print("--- Dữ liệu trả về (Markdown) ---")
                # In ra 200 ký tự đầu tiên của nội dung đã cào
                content = data.get("data", {}).get("markdown", "")
                print(content[:200] + "...")
            else:
                print(f"⚠️ API trả về lỗi logic: {data.get('error')}")
                
        elif response.status_code == 401:
            print("❌ Lỗi: API Key không hợp lệ (Unauthorized).")
        elif response.status_code == 402:
            print("❌ Lỗi: Tài khoản hết credit/gói miễn phí đã hết hạn.")
        elif response.status_code == 429:
            print("❌ Lỗi: Bạn đang gửi yêu cầu quá nhanh (Rate Limit).")
        else:
            print(f"⚠️ Mã lỗi HTTP: {response.status_code}")
            print(response.text)

    except Exception as e:
        print(f"🔥 Lỗi kết nối: {e}")

if __name__ == "__main__":
    test_firecrawl()
